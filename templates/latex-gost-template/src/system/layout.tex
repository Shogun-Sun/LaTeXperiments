% Определяет, должен ли каждый \section начинаться с нового листа.
\newif\ifConfigSectionNewPage
\ConfigSectionNewPagefalse

% Добавляет переход на новый лист перед каждым \section,
% если соответствующая настройка включена.
\AddToHook{cmd/section/before}{%
    \ifConfigSectionNewPage
    \clearpage
    \fi
}
