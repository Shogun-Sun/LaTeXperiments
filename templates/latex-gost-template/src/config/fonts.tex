\usepackage{fontspec}
\usepackage{polyglossia}

% Основной шрифт (с явным указанием пути к папке)
\setmainfont{Times_New_Roman.ttf}[
    Path          = fonts/serif/TimesNewRoman/,
    BoldFont      = Times_New_Roman_Bold.ttf,
    ItalicFont    = Times_New_Roman_Italic.ttf,
    BoldItalicFont= Times_New_Roman_Bold_Italic.ttf
]

% Кириллический шрифт для polyglossia (без принудительного Script=Cyrillic)
\newfontfamily\cyrillicfont{Times_New_Roman.ttf}[
    Path          = fonts/serif/TimesNewRoman/,
    BoldFont      = Times_New_Roman_Bold.ttf,
    ItalicFont    = Times_New_Roman_Italic.ttf,
    BoldItalicFont= Times_New_Roman_Bold_Italic.ttf
]

% Шрифты без засечек (ссылаемся на ту же папку, чтобы не зависеть от системных шрифтов)
\newfontfamily\cyrillicfontsf{Times_New_Roman.ttf}[
    Path          = fonts/serif/TimesNewRoman/,
    BoldFont      = Times_New_Roman_Bold.ttf
]

% Настройка языков
\setmainlanguage{russian}
\setotherlanguage{english}
